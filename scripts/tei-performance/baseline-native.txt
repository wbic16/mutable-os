// Qualification-only copy of 5e2f0f4 preparation, with the same stage probes.
import { decodeBlockTree } from '../../block-tree/codecs';
import { captureNative, nativeBytes } from '../../persistence/native-resource';
import { convertTei } from './convert';
import { teiLimits, type TeiLimits } from './model';
import type { SuppliedFile } from '../file-input';
export async function prepareTeiDocument(file: SuppliedFile, options: Partial<TeiLimits> = {}, profile = false) {
  const timings: Record<string, number> | undefined = profile ? {} : undefined;
  let last = profile ? performance.now() : 0;
  const stage = (name: string) => { if (timings) { const now = performance.now(); timings[name] = now - last; last = now; } };
  const limits = teiLimits(options);
  if (file.blob.size > limits.bytes) throw Error(`TEI input exceeds the configured ${limits.bytes}-byte limit. Nothing was imported.`);
  const input = new Uint8Array(await file.blob.arrayBuffer()); stage('fileRead');
  const result = convertTei(input, limits, file.name, timings); stage('conversionTotal');
  const state = decodeBlockTree(result.document).state; stage('canonicalConstruction');
  const resource = captureNative(state, result.document.id); stage('nativeCaptureAndValidation');
  const bytes = nativeBytes(resource); stage('nativeSerialization');
  const counts = profile ? { contents: Object.keys(state.contents).length, placements: Object.keys(state.placements).length, definitions: Object.keys(result.document.linkedAnnotations as object).length, blocks: Object.values(state.contents).filter(c => c.viewType !== 'text-cell' && c.viewType !== 'image-cell').length, standoffSegments: Object.values(state.contents).reduce((n,c) => n + (Array.isArray(c.payload.standoffProperties) ? c.payload.standoffProperties.length : 0), 0) } : undefined;
  return { ...(profile ? { timings, counts } : {}), bytes, id: result.document.id, diagnostics: result.diagnostics, truncated: result.truncated, stats: result.stats };
}
